### `team-test/models.py`

#### Enums

```python
from enum import Enum

class Status(str, Enum):
    TODO        = "todo"
    IN_PROGRESS = "in_progress"
    DONE        = "done"

class Priority(str, Enum):
    LOW      = "low"
    MEDIUM   = "medium"
    HIGH     = "high"
    CRITICAL = "critical"
```

Using `str, Enum` so values serialize cleanly to/from JSON without custom encoders.

#### Dataclasses

```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional

@dataclass
class Task:
    id:          str
    title:       str
    description: str               = ""
    status:      Status            = Status.TODO
    priority:    Priority          = Priority.MEDIUM
    due_date:    Optional[str]     = None   # ISO-8601 string, e.g. "2026-04-01"
    project_id:  Optional[str]     = None
    created_at:  str               = field(default_factory=lambda: datetime.utcnow().isoformat())
    updated_at:  str               = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> dict:
        """Serialize to plain dict for JSON storage."""
        ...

    @classmethod
    def from_dict(cls, data: dict) -> "Task":
        """Deserialize from plain dict loaded from JSON."""
        ...

@dataclass
class Project:
    id:          str
    name:        str
    description: str  = ""
    created_at:  str  = field(default_factory=lambda: datetime.utcnow().isoformat())

    def to_dict(self) -> dict: ...

    @classmethod
    def from_dict(cls, data: dict) -> "Project": ...
```

**Key design decisions**:
- `due_date` stored as ISO-8601 string to avoid `datetime` JSON serialization complexity.
- `status` and `priority` stored as their `.value` strings in JSON.
- `to_dict` / `from_dict` are the serialization boundary — storage layer calls these exclusively.
- `id` is a `str` (UUID4 hex) generated by the service layer, not the model itself, keeping models pure data containers.

## Phase 2 — Storage Layer (`team-test/storage.py`)

**File**: `team-test/storage.py`

### Class: `JSONFileStorage`

```python
import json, os, uuid
from pathlib import Path
from typing import Optional
from team_test.models import Task, Project   # relative import within package

class JSONFileStorage:
    DEFAULT_PATH = Path("tasks_db.json")

    def __init__(self, db_path: Path | str = DEFAULT_PATH):
        self.db_path = Path(db_path)
        self._ensure_file()

    # ── internal helpers ──────────────────────────────────────────────────
    def _ensure_file(self) -> None:
        """Create empty DB file if it doesn't exist."""

    def _load(self) -> dict:
        """Load full JSON structure: {"tasks": {...}, "projects": {...}}"""

    def _save(self, data: dict) -> None:
        """Atomically write JSON to disk (write to .tmp then rename)."""

    # ── Task CRUD ─────────────────────────────────────────────────────────
    def save_task(self, task: Task) -> Task:
        """Insert or update a Task. Returns the saved Task."""

    def get_task(self, task_id: str) -> Optional[Task]:
        """Return Task by id, or None if not found."""

    def list_tasks(
        self,
        project_id: Optional[str] = None,
        status: Optional[str] = None,
        priority: Optional[str] = None,
    ) -> list[Task]:
        """Return all tasks, optionally filtered."""

    def delete_task(self, task_id: str) -> bool:
        """Delete task by id. Returns True if deleted, False if not found."""

    # ── Project CRUD ──────────────────────────────────────────────────────
    def save_project(self, project: Project) -> Project:
        """Insert or update a Project."""

    def get_project(self, project_id: str) -> Optional[Project]:
        """Return Project by id, or None."""

    def list_projects(self) -> list[Project]:
        """Return all projects."""

    def delete_project(self, project_id: str) -> bool:
        """Delete project and optionally orphan its tasks."""
```

**Key design decisions**:
- JSON structure: `{"tasks": {"<id>": {...}}, "projects": {"<id>": {...}}}` — dict-keyed by id for O(1) lookup.
- Atomic write: write to `<path>.tmp` then `os.replace()` to avoid corruption on crash.
- `_load` / `_save` are private; all public methods go through them.
- `list_tasks` supports optional filter kwargs so the service layer can delegate filtering to storage when possible.
- Import path: `from team_test.models import Task, Project` (package name uses underscore since directory is `team-test`).

## Phase 3 — Business Logic (`team-test/services.py`)

**File**: `team-test/services.py`

### Class: `TaskService`

```python
import uuid
from datetime import datetime, date
from typing import Optional
from team_test.models import Task, Status, Priority
from team_test.storage import JSONFileStorage

class TaskService:
    def __init__(self, storage: JSONFileStorage):
        self.storage = storage

    def create_task(
        self,
        title: str,
        description: str = "",
        priority: Priority = Priority.MEDIUM,
        due_date: Optional[str] = None,
        project_id: Optional[str] = None,
    ) -> Task:
        """Generate UUID, create Task dataclass, persist via storage.save_task."""

    def get_task(self, task_id: str) -> Optional[Task]:
        """Delegate to storage.get_task."""

    def list_tasks(self, **filters) -> list[Task]:
        """Delegate to storage.list_tasks with optional filters."""

    def update_task(self, task_id: str, **kwargs) -> Optional[Task]:
        """Fetch, mutate allowed fields, update updated_at, re-save."""

    def delete_task(self, task_id: str) -> bool:
        """Delegate to storage.delete_task."""

    def change_status(self, task_id: str, new_status: Status) -> Optional[Task]:
        """Convenience wrapper around update_task(status=new_status)."""

    def assign_to_project(self, task_id: str, project_id: str) -> Optional[Task]:
        """Set task.project_id; validates project exists first."""

    def get_overdue_tasks(self) -> list[Task]:
        """Return tasks where due_date < today and status != DONE."""

    def get_tasks_by_priority(self, priority: Priority) -> list[Task]:
        """Return tasks filtered by priority."""

    def get_stats(self) -> dict:
        """Return counts by status and priority for the stats subcommand."""
```

### Class: `ProjectService`

```python
class ProjectService:
    def __init__(self, storage: JSONFileStorage):
        self.storage = storage

    def create_project(self, name: str, description: str = "") -> Project:
        """Generate UUID, create Project, persist."""

    def get_project(self, project_id: str) -> Optional[Project]:
        """Delegate to storage.get_project."""

    def list_projects(self) -> list[Project]:
        """Delegate to storage.list_projects."""

    def delete_project(self, project_id: str) -> bool:
        """Delete project; tasks with this project_id become unassigned."""

    def get_project_tasks(self, project_id: str) -> list[Task]:
        """Return all tasks belonging to a project."""
```

**Key design decisions**:
- Services own UUID generation (`uuid.uuid4().hex`) — models stay pure.
- `get_overdue_tasks` compares `task.due_date` (ISO string) to `date.today().isoformat()` using string comparison (ISO dates sort lexicographically).
- `get_stats` returns `{"by_status": {...}, "by_priority": {...}, "total": N, "overdue": N}`.
- Services accept a `storage` instance via constructor injection — makes unit testing trivial (pass a storage backed by a temp file).

1. Ensuring `team-test/__init__.py` exists (makes it a package).

## Risk & Mitigation

| Risk | Mitigation |
|---|---|
| `team-test` hyphen breaks Python imports | Use relative imports (`.models`, `.storage`) inside package; run via `python -m team_test.cli` |
| JSON file corruption on concurrent writes | Atomic write via `os.replace()` on temp file |
| `due_date` comparison edge cases | Normalize to ISO-8601 date string; document that time component is ignored |
| ANSI codes break non-TTY output | Guard with `sys.stdout.isatty()` check |
| Windows path issues | Use `pathlib.Path` throughout storage layer |
| Test isolation (shared DB state) | Each test uses `tmp_path` pytest fixture for a fresh temp file |
